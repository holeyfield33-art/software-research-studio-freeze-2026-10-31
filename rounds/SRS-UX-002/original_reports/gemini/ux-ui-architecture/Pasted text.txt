// Example: State-buffered input transition pattern
interface JourneyState {
  status: 'idle' | 'editing' | 'saving' | 'error';
  draftValue: string | null;
  canonicalValue: string | null;
}

export function draftReducer(state: JourneyState, action: { type: string; payload?: string }): JourneyState {
  switch (action.type) {
    case 'EDIT':
      return { ...state, status: 'editing', draftValue: action.payload ?? null };
    case 'COMMIT_START':
      return { ...state, status: 'saving' };
    case 'COMMIT_SUCCESS':
      return { ...state, status: 'idle', canonicalValue: state.draftValue };
    default:
      return state;
  }
}
